```js
window.kexec = async function(code, wait=15000){ const s=(await (await fetch('/api/sessions')).json())[0]; const kid=s.kernel.id; const ws=new WebSocket('ws://'+location.host+'/api/kernels/'+kid+'/channels?session_id=claude'+Date.now()); const out=[]; const mid='m'+Date.now(); await new Promise((res,rej)=>{ws.onopen=res; ws.onerror=rej;}); return await new Promise(res=>{ const t=setTimeout(()=>{ws.close(); res(out.join('')+'\n[timeout]')}, wait); ws.onmessage=e=>{const m=JSON.parse(e.data); if(!m.parent_header||m.parent_header.msg_id!==mid) return; const ty=m.header.msg_type; if(ty==='stream') out.push(m.content.text); else if(ty==='execute_result') out.push(JSON.stringify(m.content.data['text/plain'])); else if(ty==='error') out.push(m.content.ename+': '+m.content.evalue); else if(ty==='status'&&m.content.execution_state==='idle'){clearTimeout(t); ws.close(); res(out.join(''));}}; ws.send(JSON.stringify({header:{msg_id:mid,username:'claude',session:'claude',msg_type:'execute_request',version:'5.3'},parent_header:{},metadata:{},content:{code:code,silent:false,store_history:false,user_expressions:{},allow_stdin:false,stop_on_error:true},channel:'shell'})); }); };
```

## 3. 지금 설정 (config.json 주요 값)
- drive: speed 0.30, cm_s_per_unit 16.7, motor_deadband 0.05, min_motor -0.5
- ctrl: method **path**, lookahead_cm 7, min_track_motor 0.18, pivot_motor 0.35, **pivot_deg_s 15.19**, path_max_blind_s 8
- branch: yellow left / orange right / green right / purple right, **straight_after_yellow_s 0.3**
- track: rows 16, max_floor_x_cm 25, near_fit_x_cm 18, max_miss_rows 4(기본값)
- stop.mode **pass**(아직 서지 않음), loop 20Hz(한 바퀴 일 약 40~48ms)
